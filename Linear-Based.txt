//@version=6
indicator("Linear based", overlay = true, format = format.price, precision = 5)

//HMA Inputs
length = input.int(9, "Length", minval = 2, group = "HMA")
src = input(close, "Source", group = "HMA")
hullma = ta.wma(2*ta.wma(src, length/2)-ta.wma(src, length), math.floor(math.sqrt(length)))

//RSI
CurrentRSI = ta.rsi(close, 7)
RSILong = false
RSIShort = false

if CurrentRSI < 42
    RSIShort := true

else if CurrentRSI > 52
    RSILong := true

else
    RSIShort := false
    RSILong := false

//HMA  쌍봉/쌍바닥
p_leftLen  = input.int(3, "p_leftLen", group = "쌍봉/쌍바닥")
p_rightLen = input.int(3, "p_rightLen", group = "쌍봉/쌍바닥")

// 1. 피벗 고점/저점 탐지
pHigh = ta.pivothigh(hullma, p_leftLen, p_rightLen)
pLow  = ta.pivotlow(hullma, p_leftLen, p_rightLen)

// 2. 가장 최근의 피벗 가격을 저장할 변수 (새로운 피벗 전까지 유지)
var float lastPivotHigh = na
var float lastPivotLow  = na

if not na(pHigh)
    lastPivotHigh := pHigh
if not na(pLow)
    lastPivotLow := pLow

// 3. 돌파 조건 정의
// hmaLong: 현재가가 최근 피벗 고점을 상향 돌파
hmaLong  = ta.crossover(hullma, lastPivotHigh)
// hmaShort: 현재가가 최근 피벗 저점을 하향 돌파
hmaShort = ta.crossunder(hullma, lastPivotLow)



//1시간
CloseHour = request.security(syminfo.tickerid, timeframe = "60", expression = close)
OpenHour = request.security(syminfo.tickerid, timeframe = "60", expression = open)
HourLong = false
HourShort = false

if CloseHour > OpenHour
    HourLong := true
    HourShort := false
else if CloseHour < OpenHour
    HourShort := true
    HourLong := false



if hmaLong and HourLong and RSILong
    label.new(bar_index, high, "Long", color = color.blue, size = size.small)
else if hmaShort and HourShort and RSIShort
    label.new(bar_index, high, "Short", color = color.red, size = size.small)

// emaLong == true and hullLong == true and ichimokuLong == true and 
// emaShort == true and hullShort == true and ichimokuShort == true and 

plot(lastPivotHigh, "Last Pivot High (Resistance)", color = color.new(color.blue, 60), style = plot.style_linebr, linewidth = 3)
plot(lastPivotLow, "Last Pivot Low (Support)", color = color.new(color.red, 60), style = plot.style_linebr, linewidth = 3)
